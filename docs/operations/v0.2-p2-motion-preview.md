& "D:\AI-Video\.venv\Scripts\python.exe" -m py_compile .\scripts\v02_p2_multi_shot.py
& "D:\AI-Video\.venv\Scripts\python.exe" .\scripts\v02_p2_multi_shot.py --root "D:\AI-Video"
